cmake_minimum_required(VERSION 3.16)
project(AlgoRoute VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(algoroute_lib src/graph.cpp)
target_include_directories(algoroute_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(algoroute_lib PRIVATE -Wall -Wextra -Wpedantic)
add_executable(algoroute src/main.cpp)
target_link_libraries(algoroute PRIVATE algoroute_lib)
include(CTest)
if(BUILD_TESTING)
 add_executable(test_graph tests/test_graph.cpp)
 target_link_libraries(test_graph PRIVATE algoroute_lib)
 add_test(NAME graph_algorithms COMMAND test_graph)
endif()
add_executable(benchmark_algoroute benchmarks/benchmark.cpp)
target_link_libraries(benchmark_algoroute PRIVATE algoroute_lib)
